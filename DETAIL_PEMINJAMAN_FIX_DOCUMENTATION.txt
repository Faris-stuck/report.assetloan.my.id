STEP 2: Replaced Hardcoded Data with Dynamic API Calls
File: admin/peminjaman/detail-peminjaman.html

BEFORE (50+ lines of hardcoded data):
```javascript
const dataDitolak = [
  { "id": 8, "kode_peminjaman": "PMJ-004", ... },
  { "id": 9, "kode_peminjaman": "PMJ-008", ... }
];

dataDitolak.forEach(item => {
  // build HTML
});
```

AFTER (120+ lines of dynamic code):
```javascript
// 1. Get parameters from URL
const params = new URLSearchParams(window.location.search);
const kode = params.get("kode") || '';
const id = params.get("id") || '';

// 2. Fetch from database via API
let apiUrl = `${API_BASE_URL}/peminjaman/get_all.php`;
if (kode) {
    apiUrl += `?kode=${encodeURIComponent(kode)}`;
} else if (id) {
    apiUrl += `?id=${encodeURIComponent(id)}`;
}

// 3. Load data from actual database
fetch(apiUrl)
    .then(res => res.json())
    .then(res => {
        if (!res.success || !res.data || res.data.length === 0) {
            // Show error if record not found
            document.getElementById('detailContainer').innerHTML = 
                '<div class="alert alert-danger">Data not found</div>';
            return;
        }

        // Get real data from database
        const data = res.data[0];
        
        // Fetch barang details via secondary API
        return fetch(`${API_BASE_URL}/user/get-detail.php?peminjaman_id=${data.id}`)
            .then(barangRes => barangRes.json())
            .then(barangres => {
                // Build HTML with real data
                let html = `
                    <div class="detail-container">
                        <h3>${escapeHtml(data.kode_peminjaman)}</h3>
                        <p>Name: ${escapeHtml(data.nama_peminjam)}</p>
                        <p>Date: ${data.tanggal_pinjam}</p>
                        <table class="table">
                            <thead>
                                <tr>
                                    <th>Item</th>
                                    <th>Qty</th>
                                    <th>Location</th>
                                    <th>Condition</th>
                                </tr>
                            </thead>
                            <tbody>
                `;
                
                // Populate with real barang from database
                barangres.data.forEach(barang => {
                    html += `
                        <tr>
                            <td>${escapeHtml(barang.nama_barang)}</td>
                            <td>${barang.jumlah}</td>
                            <td>${escapeHtml(barang.lokasi)}</td>
                            <td>${escapeHtml(barang.kondisi_pinjam)}</td>
                        </tr>
                    `;
                });
                
                html += `
                            </tbody>
                        </table>
                    </div>
                `;
                
                // Insert into DOM
                document.getElementById('detailContainer').innerHTML = html;
            });
    })
    .catch(error => {
        // Error handling
        console.error('Error loading data:', error);
        document.getElementById('detailContainer').innerHTML = 
            '<div class="alert alert-danger">Error loading data</div>';
    });

// Security: HTML escaping function
function escapeHtml(text) {
    if (!text) return '';
    return text
        .replace(/&/g, '&amp;')
        .replace(/</g, '&lt;')
        .replace(/>/g, '&gt;')
        .replace(/"/g, '&quot;')
        .replace(/'/g, '&#039;');
}
```

Both APIs validate user role/session before returning data.
